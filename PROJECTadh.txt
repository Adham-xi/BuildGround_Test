string StarPassword()
{
    string result;
    char character;

    while ((character = _getch()) != '\r')   // Enter key
    {
        if (character == '\b')               // Backspace key
        {
            if (!result.empty())
            {
                result.pop_back();     // remove last char from the real string
                cout << "\b \b";       // erase the last '*' on screen
            }
        }
        else
        {
            result += character;              // store real char
            cout << '*';               // show a star instead
        }
    }

    cout << endl;
    return result;
}

int Receipt()
{
    float total_cost = 0;
    float tax;
    for(Orders i:selectedOrders)
    {
        total_cost = total_cost + (i.item.Price *   i.quantity );
    }
    
    tax = 14.0/100 * total_cost;
    total_cost += tax;

    
    printReceipt(total_cost,tax);
    

    int confirmation;
    cout <<"Confirm? (1-yes/2-no) : ";
    cin >> confirmation;

    if(confirmation == 1)
    {
        return paying;
    }
    else if(confirmation == 2)
    {
        selectedOrders.clear();
        return menu_user;
    }
        
    return Final_Receipt;
}

void direction()
{
    vector<int> historyStage;
    int currentStage = start;
    int callStage;

    while(currentStage != endProgram)
    {
        switch(currentStage)
        {
            case start:
                callStage = StartApp();
                break;
            case customer_interface:
                callStage = customerInterface();
                break;
            case contact:
                callStage = contactScreen();
                break;
            case eat_type:
                callStage = eatType();
                break;
            case menu_user:
                callStage = MenuScreen();
                break;
            case Final_Receipt:
                callStage = Receipt();
                break;
            case paying:
                callStage = payment();
                break;
            case owner:
                callStage = login();
                break;
            case success:
                callStage = ownerInterface();
                break;
            default:
                callStage = endProgram;
        }

        if (callStage == previous)
        {
            currentStage = historyStage.back();
            historyStage.pop_back();
        }
        else
        {
            historyStage.push_back(currentStage);
            currentStage = callStage;
        }
    }

    clearScreen();
    cout<<"Thank you for using our service!!";
}
